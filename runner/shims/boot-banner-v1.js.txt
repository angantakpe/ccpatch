// ══════════════════════════════════════════════════════════════════════════
// [PATCH] boot_banner — collect ccpatch's own boot-log lines and render them
// inside a responsive rounded box, printed once just before Claude Code's
// own banner. Other patches push lines via the order-independent contract:
//
//     (globalThis.__ccpBootBuf || (globalThis.__ccpBootBuf = [])).push([tag, msg]);
//     if (globalThis.__ccpBootKick) globalThis.__ccpBootKick();
//
// The buffer is a plain array so callers never depend on this patch having run
// first (boot injection order is non-deterministic). This installer owns the
// debounced flush: it waits until the buffer stops growing for one tick, then
// renders, so async boot lines (e.g. the fetch-before subscribe) are captured.
// Width adapts to the terminal; long lines are clipped with an ellipsis.
// ══════════════════════════════════════════════════════════════════════════
(function __ccpBootBanner() {
  if (globalThis.__ccpBootKick) return;
  var TITLE = '__CCP_BOOT_TITLE__';
  var buf = (globalThis.__ccpBootBuf || (globalThis.__ccpBootBuf = []));
  var state = { timer: null, lastLen: -1, flushed: false, printed: 0 };

  // Color is opt-out and CI-safe: honor NO_COLOR / FORCE_COLOR, else only tint
  // when stderr (where this box prints) is an interactive TTY. Self-contained —
  // this shim is injected verbatim, so it can't import runner/cli/style.mjs.
  var COLOR = (function () {
    try {
      if (process.env.NO_COLOR != null) return false;
      if (process.env.FORCE_COLOR != null && process.env.FORCE_COLOR !== '0') return true;
      return !!(process.stderr && process.stderr.isTTY);
    } catch (_) { return false; }
  })();
  function sgr(open, close, s) { return COLOR ? ('\u001b[' + open + 'm' + s + '\u001b[' + close + 'm') : String(s); }
  function cyan(s) { return sgr(36, 39, s); }
  function bold(s) { return sgr(1, 22, s); }
  function dim(s) { return sgr(2, 22, s); }
  function green(s) { return sgr(32, 39, s); }

  function cols() {
    var c = (process.stderr && process.stderr.columns) ||
            (process.stdout && process.stdout.columns) || 80;
    return c > 0 ? c : 80;
  }
  function rep(ch, n) { return n > 0 ? new Array(n + 1).join(ch) : ''; }
  function clip(s, n) {
    s = String(s == null ? '' : s);
    if (n <= 0) return '';
    if (s.length <= n) return s;
    if (n === 1) return '…';
    return s.slice(0, n - 1) + '…';
  }
  // Visible column width: emoji/astral glyphs and the sparkle occupy two
  // columns, the VS16 / ZWJ joiners occupy none. Keeps box edges aligned once
  // the title carries an emoji.
  function visLen(s) {
    s = String(s == null ? '' : s);
    var w = 0;
    for (var i = 0; i < s.length; i++) {
      var cp = s.codePointAt(i);
      if (cp > 0xFFFF) { i++; w += 2; continue; }
      if (cp === 0xFE0F || cp === 0x200D) continue;
      if (cp === 0x2728 || cp === 0x2705 || cp === 0x274C) { w += 2; continue; }
      w += 1;
    }
    return w;
  }

  function render(entries) {
    var width = Math.min(cols(), 100);
    if (width < 24) width = 24;
    var inner = width - 2;                 // visible chars between the box bars

    var tagW = 0;
    for (var i = 0; i < entries.length; i++) {
      var t = String(entries[i][0] == null ? '' : entries[i][0]);
      if (t.length > tagW) tagW = t.length;
    }
    if (tagW > 16) tagW = 16;

    var out = [];

    // Top border with embedded title: ╭─ ✨ TITLE ───────╮
    var titleSeg = ' ✨ ' + TITLE + ' ';
    if (visLen(titleSeg) > inner - 1) titleSeg = clip(titleSeg, inner - 1);
    var fill = inner - 1 - visLen(titleSeg);
    if (fill < 0) fill = 0;
    out.push(cyan('╭─') + bold(titleSeg) + cyan(rep('─', fill) + '╮'));

    // Body lines: │  tag   message            │
    for (var j = 0; j < entries.length; j++) {
      var tag = clip(entries[j][0], tagW);
      var left = '  ' + tag + rep(' ', tagW - tag.length) + '  ';
      var avail = inner - left.length;
      var msg = clip(entries[j][1], avail);
      var padN = inner - (left.length + msg.length);
      if (padN < 0) padN = 0;
      var body = '  ' + green(tag) + rep(' ', tagW - tag.length) + '  ' + dim(msg) + rep(' ', padN);
      out.push(cyan('│') + body + cyan('│'));
    }

    // Bottom border: ╰───────────╯
    out.push(cyan('╰' + rep('─', inner) + '╯'));

    return '\n' + out.join('\n') + '\n\n';
  }

  function flush() {
    state.timer = null;
    if (state.flushed) {
      // Late arrivals (after the box was drawn) print as plain indented lines.
      for (; state.printed < buf.length; state.printed++) {
        var e = buf[state.printed];
        try { process.stderr.write('  ' + green(String(e[0])) + '  ' + dim(String(e[1])) + '\n'); } catch (_) {}
      }
      return;
    }
    // Debounce: if the buffer is still growing, wait one more tick so async
    // boot lines land in the same box.
    if (buf.length !== state.lastLen) {
      state.lastLen = buf.length;
      state.timer = setImmediate(flush);
      return;
    }
    if (buf.length === 0) return;
    state.flushed = true;
    state.printed = buf.length;
    try { process.stderr.write(render(buf)); } catch (_) {}
  }

  function kick() {
    if (!state.timer) state.timer = setImmediate(flush);
  }

  globalThis.__ccpBootKick = kick;
  // Drain anything buffered before this installer ran.
  kick();
})();

// [PATCH] boot_banner / diagnostics sink — always-on, file-backed record of
// swallowed errors. Other injected code (e.g. the fetch bus warn helper)
// appends one-line records via globalThis.__ccpDiag(tag, msg); nothing here
// ever writes to stderr in a healthy boot. Location: $CCPATCH_DIAG_DIR if
// set, else ~/.ccpatch/diagnostics (cwd-independent). ~1MB size guard:
// rotate to runtime.log.1, falling back to truncate.
//
// Also installs early LOG-ONLY process observers so the next swallowed boot
// failure is visible in the log:
//   - 'uncaughtExceptionMonitor': observational by Node design — it never
//     changes crash semantics.
//   - 'unhandledRejection': adding a listener WOULD suppress Node's default
//     (crash, exit 1). Semantics are preserved with a sole-listener rethrow:
//     if at emit time this listener is the ONLY one registered (the window
//     before the bundle installs its own handlers), the reason is rethrown,
//     which crashes the process with the error + stack and exit code 1 —
//     the same outcome Node's default --unhandled-rejections=throw mode
//     produces. When any other listener exists (the bundle installs its own
//     'unhandledRejection' handler during startup — verified in
//     cli v2.1.175), this listener only logs and the bundle's handler keeps
//     deciding the outcome, exactly as without ccpatch. Known limitation:
//     a user who explicitly launches node with --unhandled-rejections=warn|
//     none gets a crash in that pre-bundle window where they'd have gotten
//     a warning; Claude Code's own launchers never set that flag.
// ══════════════════════════════════════════════════════════════════════════
(function __ccpDiagSink() {
  try {
    if (globalThis.__ccpDiagSink_v1) return;
    globalThis.__ccpDiagSink_v1 = true;

    // Builtin access that works in every injection context: ESM output
    // (process.getBuiltinModule / __hm_nativeRequire from the esm-compat
    // header) and CJS output (plain require).
    function _mod(id) {
      try { if (typeof process.getBuiltinModule === 'function') return process.getBuiltinModule(id); } catch (_) {}
      try { if (typeof __hm_nativeRequire === 'function') return __hm_nativeRequire(id); } catch (_) {}
      try { if (typeof require === 'function') return require(id); } catch (_) {}
      return null;
    }

    var MAX_BYTES = 1024 * 1024; // ~1MB rotate threshold

    function diagFile() {
      var path = _mod('path'); var os = _mod('os');
      if (!path || !os) return null;
      var dir = '';
      try { dir = process.env.CCPATCH_DIAG_DIR || ''; } catch (_) {}
      if (!dir) dir = path.join(os.homedir(), '.ccpatch', 'diagnostics');
      return path.join(dir, 'runtime.log');
    }

    var sink = function __ccpDiagWrite(tag, msg) {
      try {
        var fs = _mod('fs'); var path = _mod('path');
        if (!fs || !path) return;
        var file = diagFile();
        if (!file) return;
        try { fs.mkdirSync(path.dirname(file), { recursive: true }); } catch (_) {}
        try {
          var st = fs.statSync(file);
          if (st && st.size > MAX_BYTES) {
            try { fs.renameSync(file, file + '.1'); }
            catch (_) { try { fs.truncateSync(file, 0); } catch (_2) {} }
          }
        } catch (_) {}
        var body = String(msg == null ? '' : msg).replace(/\s*\n\s*/g, ' | ').slice(0, 2000);
        fs.appendFileSync(file,
          new Date().toISOString() + ' [' + String(tag) + '] pid=' + process.pid + ' ' + body + '\n');
      } catch (_) {}
    };
    try { sink.where = diagFile; } catch (_) {}
    if (!globalThis.__ccpDiag) globalThis.__ccpDiag = sink;

    function describe(err) {
      try { return (err && (err.stack || err.message)) || String(err); }
      catch (_) { return '<unprintable>'; }
    }

    try {
      process.on('unhandledRejection', function __ccpDiagOnRejection(reason) {
        try {
          var m = describe(reason);
          globalThis.__ccpDiag('unhandled-rejection', m);
          // CCPATCH_PARANOID=1 keeps its loud-stderr contract.
          try {
            if (process.env.CCPATCH_PARANOID === '1') console.error('[ccp:diag][paranoid] unhandledRejection: ' + m);
          } catch (_) {}
        } catch (_) {}
        // Sole-listener rethrow: preserve Node's default crash when nothing
        // else handles the event (see header comment). Intentionally outside
        // try/catch — the throw IS the preserved default behavior.
        if (process.listenerCount('unhandledRejection') === 1) throw reason;
      });
    } catch (_) {}

    try {
      process.on('uncaughtExceptionMonitor', function __ccpDiagOnException(err) {
        try { globalThis.__ccpDiag('uncaught-exception', describe(err)); } catch (_) {}
      });
    } catch (_) {}
  } catch (_) {}
})();

// [PATCH] boot_banner / boot watchdog — one-shot, unref'd hang beacon.
//
// Failure mode this exists for: the patched CLI prints its boot banner and
// then sits silent forever (a boot-time deadlock with every error swallowed).
// ~15s after boot, if the process still hasn't produced substantive UI
// output, ONE diagnostic block is written to stderr: active handle/request
// constructor counts, child-process count, and actionable env hints.
//
// "UI rendered" proxy — cumulative bytes written to process.stdout after this
// shim runs (the boot banner itself goes to stderr and is not counted):
//   - A hung boot emits only terminal capability queries / cursor probes on
//     stdout: tens of bytes, well under 1KB (measured 0 bytes on the real
//     v2.1.175 bundle hung headless).
//   - A rendered interactive frame (Ink TUI: welcome box, trust dialog,
//     prompt) is several KB with the alternate screen ON: Ink repaints
//     continuously — ~5KB cumulative stdout within seconds of a healthy boot
//     on a default 80x24 PTY. But with CLAUDE_CODE_DISABLE_ALTERNATE_SCREEN=1
//     (which this very watchdog recommends, and which a Node boot needs) Ink
//     writes ONE static frame and only repaints on change: a healthy idle
//     boot measured ~787B by 7s and ~1.4KB by 15s — legitimately under the
//     old 2048 threshold, which made the beacon fire over a fully-rendered,
//     interactive UI (the alt-screen-off path was never accounted for).
//     The threshold sits at 512: above any hung boot's query traffic (a hung
//     headless v2.1.175 emitted 0 bytes on stdout; terminal capability probes
//     are tens of bytes) and well under the smallest healthy render measured
//     in either mode, biased toward never firing on a slow-but-healthy boot.
// The beacon is written with fs.writeSync(2, …), NOT process.stderr.write:
// once the TUI is up the bundle (Ink patch-console) replaces stderr.write
// and swallows it — a direct fd-2 write bypasses that (verified against the
// real v2.1.175 bundle under a PTY).
// False-positive containment: the watchdog only arms when an interactive UI
// is actually expected — stdout is a TTY and argv carries no recognized
// non-interactive mode (-p/--print, --version, --help, mcp, install, …).
// Piped/print/one-shot runs that legitimately stay under 4KB never arm it.
// The timer is unref'd, so it never keeps a finished process alive, and the
// stdout wrap restores itself as soon as the threshold is crossed (or the
// timer fires). Opt-out: CCPATCH_NO_WATCHDOG=1.
// Test hooks: CCPATCH_WATCHDOG_FORCE=1 (skip the interactivity gate),
// CCPATCH_WATCHDOG_MS, CCPATCH_WATCHDOG_THRESHOLD.
// ══════════════════════════════════════════════════════════════════════════
(function __ccpWatchdog() {
  try {
    if (globalThis.__ccpWatchdog_v1) return;
    globalThis.__ccpWatchdog_v1 = true;

    var env = process.env || {};
    if (env.CCPATCH_NO_WATCHDOG === '1') return;

    var force = env.CCPATCH_WATCHDOG_FORCE === '1';
    if (!force) {
      if (!process.stdout || !process.stdout.isTTY) return; // no UI expected
      var argv = (process.argv || []).slice(2);
      var NON_INTERACTIVE = ['-p', '--print', '--version', '-v', '--help', '-h',
        'mcp', 'install', 'update', 'doctor', 'migrate-installer', 'setup-token', 'api'];
      for (var i = 0; i < argv.length; i++) {
        if (NON_INTERACTIVE.indexOf(argv[i]) !== -1) return;
      }
    }

    var delayMs = parseInt(env.CCPATCH_WATCHDOG_MS, 10);
    if (!(delayMs > 0)) delayMs = 15000;
    var threshold = parseInt(env.CCPATCH_WATCHDOG_THRESHOLD, 10);
    if (!(threshold > 0)) threshold = 512;

    var bytes = 0;
    var timer = null;
    var origWrite = process.stdout.write;
    var wrapped = function __ccpWatchdogStdoutCount(chunk) {
      try {
        bytes += (typeof chunk === 'string') ? Buffer.byteLength(chunk)
          : (chunk && chunk.length) || 0;
        if (bytes >= threshold) { restore(); disarm(); } // healthy — stand down early
      } catch (_) { try { restore(); } catch (_2) {} }
      return origWrite.apply(process.stdout, arguments);
    };
    function restore() {
      try { if (process.stdout.write === wrapped) process.stdout.write = origWrite; } catch (_) {}
    }
    function disarm() {
      try { if (timer) { clearTimeout(timer); timer = null; } } catch (_) {}
    }

    function tally(list) {
      var counts = {};
      for (var i = 0; i < list.length; i++) {
        var n = 'unknown';
        try { n = (list[i] && list[i].constructor && list[i].constructor.name) || 'unknown'; } catch (_) {}
        counts[n] = (counts[n] || 0) + 1;
      }
      var parts = [];
      for (var k in counts) parts.push(k + ' x' + counts[k]);
      return parts.length ? parts.join(', ') : '(none)';
    }

    function beacon() {
      timer = null;
      restore();
      if (bytes >= threshold) return; // UI rendered — stay silent
      var handles = [];
      var requests = [];
      try { if (typeof process._getActiveHandles === 'function') handles = process._getActiveHandles() || []; } catch (_) {}
      try { if (typeof process._getActiveRequests === 'function') requests = process._getActiveRequests() || []; } catch (_) {}
      var children = 0;
      for (var i = 0; i < handles.length; i++) {
        try { if (handles[i] && handles[i].constructor && handles[i].constructor.name === 'ChildProcess') children++; } catch (_) {}
      }
      var where = '';
      try { where = (globalThis.__ccpDiag && globalThis.__ccpDiag.where) ? (globalThis.__ccpDiag.where() || '') : ''; } catch (_) {}
      var altScreen;
      try { altScreen = env.CLAUDE_CODE_DISABLE_ALTERNATE_SCREEN; } catch (_) {}
      var lines = [
        '',
        '[ccpatch] boot watchdog: still no substantive UI output ' + Math.round(delayMs / 1000) + 's after boot',
        '  stdout bytes since boot: ' + bytes + ' (render threshold: ' + threshold + ')',
        '  active handles:  ' + tally(handles),
        '  active requests: ' + tally(requests),
        '  child processes: ' + children,
        '  hints:',
      ];
      if (!(process.versions && process.versions.bun)) {
        var altOn = altScreen === '1' ||
          (typeof altScreen === 'string' && altScreen.toLowerCase() === 'true');
        if (altOn) {
          // Already disabled — telling the user to set what's already set is
          // self-contradictory noise. With alt-screen off a healthy Node boot
          // renders a single static frame, so low stdout volume is EXPECTED
          // and not itself the fault; report the state, don't prescribe it.
          lines.push('    - running under Node with CLAUDE_CODE_DISABLE_ALTERNATE_SCREEN=' +
            JSON.stringify(altScreen) + ' (alt-screen already disabled; a static single-frame render is normal)');
        } else {
          lines.push('    - running under Node: fullscreen TUI unavailable — CLAUDE_CODE_DISABLE_ALTERNATE_SCREEN should be set (currently: ' +
            (altScreen === undefined ? '<unset>' : JSON.stringify(altScreen)) + ')');
        }
      }
      if (where) lines.push('    - check the ccpatch diagnostics log for swallowed errors: ' + where);
      lines.push('    - relaunch with CCPATCH_PARANOID=1 to surface swallowed subscriber errors on stderr');
      lines.push('    - set CCPATCH_NO_WATCHDOG=1 to silence this beacon');
      var block = lines.join('\n') + '\n';
      // Direct fd-2 write: the bundle's TUI replaces process.stderr.write and
      // swallows it once rendering starts — and a hung boot may still have
      // gotten that far. fs.writeSync(2, …) is not interceptable from JS.
      var wrote = false;
      try {
        var fs = null;
        try { if (typeof process.getBuiltinModule === 'function') fs = process.getBuiltinModule('fs'); } catch (_) {}
        try { if (!fs && typeof __hm_nativeRequire === 'function') fs = __hm_nativeRequire('fs'); } catch (_) {}
        try { if (!fs && typeof require === 'function') fs = require('fs'); } catch (_) {}
        if (fs && typeof fs.writeSync === 'function') { fs.writeSync(2, block); wrote = true; }
      } catch (_) {}
      if (!wrote) { try { process.stderr.write(block); } catch (_) {} }
      try { if (globalThis.__ccpDiag) globalThis.__ccpDiag('watchdog', 'beacon fired: stdout bytes=' + bytes + ' handles=' + tally(handles) + ' requests=' + tally(requests) + ' children=' + children); } catch (_) {}
    }

    process.stdout.write = wrapped;
    timer = setTimeout(function () { try { beacon(); } catch (_) { try { restore(); } catch (_2) {} } }, delayMs);
    // MUST never keep an otherwise-finished process alive.
    if (timer && typeof timer.unref === 'function') timer.unref();
  } catch (_) {}
})();
